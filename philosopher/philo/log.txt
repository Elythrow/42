// Fixed print_status function (time.c)
void print_status(t_philosopher *philo, char *action)
{
    pthread_mutex_lock(&philo->table->death_lock);
    if (!philo->table->simulation_stop)
    {
        pthread_mutex_unlock(&philo->table->death_lock);
        
        pthread_mutex_lock(&philo->table->print_lock);
        printf("%lld %d %s\n", 
            get_time_ms() - philo->table->start_time,
            philo->id,
            action);
        pthread_mutex_unlock(&philo->table->print_lock);
    }
    else
        pthread_mutex_unlock(&philo->table->death_lock);
}

// Fixed all_philosophers_satisfied function (monitor.c)
static bool all_philosophers_satisfied(t_table *table)
{
    int i;
    int meals;
    
    if (table->required_meals == -1)
        return (false);
    i = 0;
    while (i < table->philo_count)
    {
        pthread_mutex_lock(&table->philosophers[i]->meal_lock);
        meals = table->philosophers[i]->meals_eaten;
        pthread_mutex_unlock(&table->philosophers[i]->meal_lock);
        if (meals < table->required_meals)
            return (false);
        i++;
    }
    return (true);
}

Additional Recommendations:

Fork initialization race: Make sure all forks are initialized before any philosopher thread starts.The barrier solution mentioned above would help with this.
Memory barriers: Some compilers might reorder operations. Consider using pthread_mutex_lock even for simple boolean checks to ensure proper memory synchronization.
Double-check pattern: In some places, you're using a pattern where you check a condition, lock, then check again. This is good but make sure the first check is also protected or use atomic operations.
Consistent lock ordering: You're already doing this well with the even/odd philosopher fork ordering, which prevents deadlock.
